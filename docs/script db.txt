-- BASE DE DATOS: SISTEMA DE FACTURACIÓN
-- PostgreSQL
-- Version 3:
--   - clientes: se quita contrato_ruta (el contrato se maneja
--     físico/local, fuera del sistema); contacto_emergencia
--     pasa a ser solo el número de celular, y se agrega
--     "referencia" para la relación con el cliente (ej. Mamá)
--   - productos: se deja solo con nombre y precio (sin
--     categoria ni stock; no se está pidiendo ese control)
--   - corregidas comas colgantes que rompían el CREATE TABLE
--     de clientes y productos
--   - corregido typo 'FEMENIO' -> 'FEMENINO' en el CHECK de
--     genero
-- ============================================================


-- ============================================================
-- 1. CLIENTES
-- ============================================================

CREATE TABLE clientes (
    id_cliente SERIAL PRIMARY KEY,

    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,

    cedula VARCHAR(20) NOT NULL UNIQUE,

    fecha_nacimiento DATE,

    correo VARCHAR(150),
    telefono VARCHAR(30),

    genero VARCHAR(30),

    contacto_emergencia VARCHAR(150),
    contrato_ruta VARCHAR(150),
    vigente_hasta DATE,

    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    cedula VARCHAR(10) NOT NULL UNIQUE,
    fecha_nacimiento DATE,
    correo VARCHAR(150),
    telefono VARCHAR(10),
    genero VARCHAR(30),
    contacto_emergencia VARCHAR(10),
    referencia VARCHAR(50),
    vigente_hasta DATE,
    fecha_registro TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT chk_genero
        CHECK (genero IN ('MASCULINO', 'FEMENINO', 'OTRO'))
);


-- ============================================================
-- 2. EMPLEADOS
-- ============================================================

CREATE TABLE empleados (
    id_empleado SERIAL PRIMARY KEY,

    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,

    usuario VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,

    rol VARCHAR(30) NOT NULL,

    activo BOOLEAN NOT NULL DEFAULT TRUE,

    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    usuario VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL, -- pendiente: se asume hash (bcrypt), no texto plano
    rol VARCHAR(30) NOT NULL,
    activo BOOLEAN NOT NULL DEFAULT TRUE,
    fecha_registro TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT chk_empleado_rol
        CHECK (rol IN ('ADMIN', 'SUPERVISOR', 'VENDEDOR'))
);


-- ============================================================
-- 3. PRODUCTOS
-- ============================================================
-- Solo artículos físicos que se venden (alquiler, tienda).
-- Sin categoria ni stock: no se ha pedido ese control por
-- ahora. Las membresías viven aparte, en planes_membresia.

CREATE TABLE productos (
    id_producto SERIAL PRIMARY KEY,

    nombre VARCHAR(150) NOT NULL,
    categoria VARCHAR(100) NOT NULL,
    tipo VARCHAR(50) NOT NULL,

    precio NUMERIC(10,2) NOT NULL,
    stock INTEGER NOT NULL DEFAULT 0,

    es_membresia BOOLEAN NOT NULL DEFAULT FALSE,

    tipo_membresia VARCHAR(50),
    dias_incluidos INTEGER,

    CONSTRAINT chk_producto_precio
        CHECK (precio >= 0),

    CONSTRAINT chk_producto_stock
        CHECK (stock >= 0),

    CONSTRAINT chk_producto_dias
        CHECK (dias_incluidos IS NULL OR dias_incluidos > 0),

    CONSTRAINT chk_producto_membresia
        CHECK (
            (es_membresia = FALSE
                AND tipo_membresia IS NULL
                AND dias_incluidos IS NULL)
            OR
            (es_membresia = TRUE
                AND tipo_membresia IS NOT NULL
                AND dias_incluidos IS NOT NULL)
        )
);


-- ============================================================
-- 4. FACTURAS
    nombre VARCHAR(150) NOT NULL,
    precio NUMERIC(10,2) NOT NULL,

    CONSTRAINT chk_producto_precio
        CHECK (precio >= 0)
);


-- ============================================================
-- 4. PLANES DE MEMBRESÍA
-- ============================================================

CREATE TABLE planes_membresia (
    id_plan SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL UNIQUE,
    tipo VARCHAR(20) NOT NULL,
    precio NUMERIC(10,2) NOT NULL,
    duracion INTEGER NOT NULL,
    unidad_duracion VARCHAR(10) NOT NULL,
    permite_pareja BOOLEAN NOT NULL DEFAULT FALSE,
    permite_descuento BOOLEAN NOT NULL DEFAULT FALSE,
    activo BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT chk_plan_tipo
        CHECK (tipo IN ('JUNIOR', 'SENIOR', 'PASES')),

    CONSTRAINT chk_plan_precio
        CHECK (precio >= 0),

    CONSTRAINT chk_plan_duracion
        CHECK (duracion > 0),

    CONSTRAINT chk_plan_unidad
        CHECK (unidad_duracion IN ('DIA', 'MES'))
);


-- ============================================================
-- 5. FACTURAS
-- ============================================================

CREATE TABLE facturas (
    id_factura SERIAL PRIMARY KEY,

    numero_factura VARCHAR(30) NOT NULL UNIQUE,

    id_cliente INTEGER NOT NULL,
    id_empleado INTEGER NOT NULL,

    fecha TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    metodo_pago VARCHAR(30) NOT NULL,

    descuento_porcentaje NUMERIC(5,2) NOT NULL DEFAULT 0,

    subtotal NUMERIC(10,2) NOT NULL,
    iva NUMERIC(10,2) NOT NULL,
    total NUMERIC(10,2) NOT NULL,

    estado VARCHAR(30) NOT NULL DEFAULT 'EMITIDA',

    motivo_anulacion VARCHAR(255),

    CONSTRAINT fk_factura_cliente
        FOREIGN KEY (id_cliente)
        REFERENCES clientes(id_cliente),

    CONSTRAINT fk_factura_empleado
        FOREIGN KEY (id_empleado)
        REFERENCES empleados(id_empleado),

    CONSTRAINT chk_factura_metodo_pago
        CHECK (metodo_pago IN ('EFECTIVO', 'TRANSFERENCIA')),

    CONSTRAINT chk_factura_descuento
        CHECK (
            descuento_porcentaje >= 0
            AND descuento_porcentaje <= 100
        ),

    CONSTRAINT chk_factura_subtotal
        CHECK (subtotal >= 0),

    CONSTRAINT chk_factura_iva
        CHECK (iva >= 0),

    CONSTRAINT chk_factura_total
        CHECK (total >= 0),

    CONSTRAINT chk_factura_estado
        CHECK (
            estado IN (
                'EMITIDA',
                'ANULADA'
            )
            estado IN ('EMITIDA', 'ANULADA')
        ),

    CONSTRAINT chk_factura_anulacion
        CHECK (
            (estado = 'ANULADA' AND motivo_anulacion IS NOT NULL)
            OR
            (estado <> 'ANULADA')
        )
);


-- ============================================================
-- 5. DETALLE DE FACTURA
-- 6. DETALLE DE FACTURA
-- ============================================================

CREATE TABLE detalle_factura (
    id_detalle SERIAL PRIMARY KEY,

    id_factura INTEGER NOT NULL,
    id_producto INTEGER NOT NULL,

    cantidad INTEGER NOT NULL,

    precio_unitario NUMERIC(10,2) NOT NULL,
    subtotal NUMERIC(10,2) NOT NULL,

    CONSTRAINT fk_detalle_factura
        FOREIGN KEY (id_factura)
        REFERENCES facturas(id_factura)
        ON DELETE CASCADE,

    CONSTRAINT fk_detalle_producto
        FOREIGN KEY (id_producto)
        REFERENCES productos(id_producto),

    CONSTRAINT chk_detalle_cantidad
        CHECK (cantidad > 0),

    CONSTRAINT chk_detalle_precio
        CHECK (precio_unitario >= 0),

    CONSTRAINT chk_detalle_subtotal
        CHECK (subtotal >= 0)
);


-- ============================================================
-- 7. MEMBRESÍAS
-- ============================================================

CREATE TABLE membresias (
    id_membresia SERIAL PRIMARY KEY,

    id_cliente INTEGER NOT NULL,
    id_plan INTEGER NOT NULL,
    id_factura INTEGER NOT NULL,

    fecha_inicio DATE NOT NULL,
    fecha_fin DATE NOT NULL,

    -- Solo aplican a planes tipo "Por día" (PASES); NULL para
    -- planes mensuales Junior/Senior.
    dias_totales INTEGER,
    dias_restantes INTEGER,

    estado VARCHAR(30) NOT NULL DEFAULT 'ACTIVA',

    id_membresia_pareja INTEGER UNIQUE,

    CONSTRAINT fk_membresia_cliente
        FOREIGN KEY (id_cliente)
        REFERENCES clientes(id_cliente),

    CONSTRAINT fk_membresia_plan
        FOREIGN KEY (id_plan)
        REFERENCES planes_membresia(id_plan),

    CONSTRAINT fk_membresia_factura
        FOREIGN KEY (id_factura)
        REFERENCES facturas(id_factura),

    CONSTRAINT fk_membresia_pareja
        FOREIGN KEY (id_membresia_pareja)
        REFERENCES membresias(id_membresia),

    CONSTRAINT chk_membresia_fechas
        CHECK (fecha_fin >= fecha_inicio),

    CONSTRAINT chk_membresia_dias
        CHECK (dias_totales IS NULL OR dias_totales > 0),

    CONSTRAINT chk_membresia_dias_restantes
        CHECK (
            dias_restantes IS NULL
            OR (
                dias_restantes >= 0
                AND dias_restantes <= dias_totales
            )
        ),

    CONSTRAINT chk_membresia_estado
        CHECK (
            estado IN ('ACTIVA', 'VENCIDA', 'CANCELADA')
        ),

    CONSTRAINT chk_membresia_no_auto_pareja
        CHECK (
            id_membresia_pareja IS NULL
            OR id_membresia_pareja <> id_membresia
        )
);


-- ============================================================
-- 8. CATEGORÍAS DE EVENTOS
-- ============================================================

CREATE TABLE categorias_evento (
    id_categoria SERIAL PRIMARY KEY,
    nombre VARCHAR(225) NOT NULL UNIQUE,
    color VARCHAR(30) NOT NULL
);


-- ============================================================
-- 9. EVENTOS
-- ============================================================
-- Nota: no se agregan columnas nullable para "evento de todo
-- el dia". Convención acordada: hora_inicio = 00:00:00 y
-- hora_fin = 23:59:59, e indicar "(todo el dia)" en el titulo
-- al crear el evento.

CREATE TABLE eventos (
    id_evento SERIAL PRIMARY KEY,

    titulo VARCHAR(150) NOT NULL,
    id_categoria INTEGER NOT NULL,

    fecha DATE NOT NULL,
    hora_inicio TIME NOT NULL,
    hora_fin TIME NOT NULL,

    descripcion TEXT,
    cupo_maximo INTEGER,

    CONSTRAINT fk_evento_categoria
        FOREIGN KEY (id_categoria)
        REFERENCES categorias_evento(id_categoria),

    CONSTRAINT chk_evento_horas
        CHECK (hora_fin > hora_inicio),

    CONSTRAINT chk_evento_cupo
        CHECK (
            cupo_maximo IS NULL
            OR cupo_maximo > 0
        )
);
